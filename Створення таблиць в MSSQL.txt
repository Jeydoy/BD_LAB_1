Створення таблиць в MSSQL

CREATE TABLE Roles (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    Name NVARCHAR(50) NOT NULL
);

CREATE TABLE Users (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    RoleId INT NOT NULL FOREIGN KEY REFERENCES Roles(Id),
    FirstName NVARCHAR(100) NOT NULL,
    LastName NVARCHAR(100) NOT NULL,
    Email NVARCHAR(255),
    UpdatedAt DATETIME2 NULL,
    UpdatedBy INT NULL FOREIGN KEY REFERENCES Users(Id),
    IsDeleted BIT NOT NULL DEFAULT 0
);

CREATE TABLE Locations (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    Name NVARCHAR(100) NOT NULL,
    Address NVARCHAR(255) NOT NULL
);


CREATE TABLE VehicleCategories (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    Name NVARCHAR(50) NOT NULL,
    BaseDailyRate DECIMAL(18,2) NOT NULL
);

CREATE TABLE VehicleModels (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    CategoryId INT NOT NULL FOREIGN KEY REFERENCES VehicleCategories(Id),
    Brand NVARCHAR(100) NOT NULL,
    Model NVARCHAR(100) NOT NULL,
    Year INT NOT NULL,
    Transmission NVARCHAR(50) NOT NULL
);

CREATE TABLE Vehicles (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    ModelId INT NOT NULL FOREIGN KEY REFERENCES VehicleModels(Id),
    LocationId INT NOT NULL FOREIGN KEY REFERENCES Locations(Id),
    VIN NVARCHAR(50) NOT NULL UNIQUE,
    LicensePlate NVARCHAR(20) NOT NULL UNIQUE,
    Mileage INT NOT NULL,
    Status NVARCHAR(50) NOT NULL, 
    UpdatedAt DATETIME2 NULL,
    UpdatedBy INT NULL FOREIGN KEY REFERENCES Users(Id),
    IsDeleted BIT NOT NULL DEFAULT 0
);


CREATE TABLE Clients (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    FirstName NVARCHAR(100) NOT NULL,
    LastName NVARCHAR(100) NOT NULL,
    Phone NVARCHAR(20) NOT NULL,
    Email NVARCHAR(255),
    Rating INT NULL,
    UpdatedAt DATETIME2 NULL,
    UpdatedBy INT NULL FOREIGN KEY REFERENCES Users(Id),
    IsDeleted BIT NOT NULL DEFAULT 0
);

CREATE TABLE ClientDocuments (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    ClientId INT NOT NULL FOREIGN KEY REFERENCES Clients(Id),
    PassportNumber NVARCHAR(50) NOT NULL,
    LicenseNumber NVARCHAR(50) NOT NULL,
    LicenseExpiryDate DATE NOT NULL
);


CREATE TABLE RentalStatuses (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    Name NVARCHAR(50) NOT NULL
);

CREATE TABLE Rentals (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    ClientId INT NOT NULL FOREIGN KEY REFERENCES Clients(Id),
    VehicleId INT NOT NULL FOREIGN KEY REFERENCES Vehicles(Id),
    PickUpLocationId INT NOT NULL FOREIGN KEY REFERENCES Locations(Id),
    DropOffLocationId INT NOT NULL FOREIGN KEY REFERENCES Locations(Id),
    ProcessedByUserId INT NOT NULL FOREIGN KEY REFERENCES Users(Id),
    StatusId INT NOT NULL FOREIGN KEY REFERENCES RentalStatuses(Id),
    StartDate DATETIME2 NOT NULL,
    EndDate DATETIME2 NOT NULL,
    ActualEndDate DATETIME2 NULL,
    UpdatedAt DATETIME2 NULL,
    UpdatedBy INT NULL FOREIGN KEY REFERENCES Users(Id),
    IsDeleted BIT NOT NULL DEFAULT 0
);

CREATE TABLE Payments (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    RentalId INT NOT NULL FOREIGN KEY REFERENCES Rentals(Id),
    Amount DECIMAL(18,2) NOT NULL,
    PaymentDate DATETIME2 NOT NULL,
    PaymentMethod NVARCHAR(50) NOT NULL 
);

CREATE TABLE AdditionalOptions (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    Name NVARCHAR(100) NOT NULL,
    PricePerDay DECIMAL(18,2) NOT NULL
);

CREATE TABLE RentalOptions (
    RentalId INT NOT NULL FOREIGN KEY REFERENCES Rentals(Id),
    OptionId INT NOT NULL FOREIGN KEY REFERENCES AdditionalOptions(Id),
    PRIMARY KEY (RentalId, OptionId) 
);


CREATE TABLE DamageReports (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    RentalId INT NULL FOREIGN KEY REFERENCES Rentals(Id), 
    VehicleId INT NOT NULL FOREIGN KEY REFERENCES Vehicles(Id),
    ReportedByUserId INT NOT NULL FOREIGN KEY REFERENCES Users(Id),
    Description NVARCHAR(MAX) NOT NULL,
    DamageDate DATETIME2 NOT NULL,
    UpdatedAt DATETIME2 NULL,
    UpdatedBy INT NULL FOREIGN KEY REFERENCES Users(Id)
);

CREATE TABLE MaintenanceRecords (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    VehicleId INT NOT NULL FOREIGN KEY REFERENCES Vehicles(Id),
    ServiceType NVARCHAR(100) NOT NULL,
    Cost DECIMAL(18,2) NOT NULL,
    ServiceDate DATETIME2 NOT NULL
);
